set(ota_sources "src/manifest.c" "src/transfer.c" "src/identity.c" "src/remote.c")
if(CONFIG_PET_OTA_ENABLED)
    list(APPEND ota_sources "src/ota_esp.c")
endif()
idf_component_register(SRCS ${ota_sources} INCLUDE_DIRS "include"
    REQUIRES "contracts" PRIV_REQUIRES "cjson" "mbedtls" "esp_http_client" "app_update"
        "esp_partition" "storage" "svc_config" "svc_growth" "ui_lvgl" "svc_ai_agent" "esp_timer")
idf_build_get_property(pet_build_dir BUILD_DIR)
set(model_args "")
set(model_deps "")
if(CONFIG_PET_AI_ESP_SR_ENABLED)
    set(model_deps "${pet_build_dir}/pet_sr/pet_sr_model_info.h")
    set(model_args --model-header "${model_deps}")
    add_dependencies(${COMPONENT_LIB} pet_sr_model_check)
endif()
set(identity_header "${CMAKE_CURRENT_BINARY_DIR}/pet_build_identity.h")
set(identity_script "${CMAKE_CURRENT_LIST_DIR}/../../../tools/build_identity.py")
set(partition_source "${CMAKE_CURRENT_LIST_DIR}/../../partitions/16m.csv")
add_custom_command(OUTPUT "${identity_header}"
    COMMAND ${PYTHON} "${identity_script}" ${model_args} --partition-table "${partition_source}" --output "${identity_header}"
    DEPENDS "${identity_script}" "${partition_source}" ${model_deps} VERBATIM)
add_custom_target(pet_identity_check DEPENDS "${identity_header}")
add_dependencies(${COMPONENT_LIB} pet_identity_check)
target_include_directories(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(CONFIG_PET_OTA_ENABLED)
    if(NOT CONFIG_BOOTLOADER_APP_ROLLBACK_ENABLE)
        message(FATAL_ERROR "PET OTA requires a USB-installed rollback-enabled bootloader")
    endif()
    if(NOT EXISTS "${CONFIG_PET_OTA_PUBLIC_KEY_DER}")
        message(FATAL_ERROR "Set PET_OTA_PUBLIC_KEY_DER to your RSA-3072 PKCS#1 public DER file")
    endif()
    target_add_binary_data(${COMPONENT_LIB} "${CONFIG_PET_OTA_PUBLIC_KEY_DER}" BINARY RENAME_TO ota_public_key)
endif()
